PASSO 4: Criar middleware: lib/auth-middleware.ts

/* ============================================================================
   lib/auth-middleware.ts
   O que é: Middleware para validar rotas autenticadas
   Onde é usado: middleware.ts (Next.js)
   Depende de: lib/auth.ts
   ============================================================================ */

import { auth } from "@/lib/auth";
import { NextResponse, type NextRequest } from "next/server";

/**
 * Middleware para validar autenticação em rotas protegidas
 * @param request NextRequest
 */
export async function withAuth(request: NextRequest) {
  const session = await auth();
  
  // Se não tem sessão e tenta acessar rota protegida
  if (!session && request.nextUrl.pathname.startsWith("/painel")) {
    return NextResponse.redirect(new URL("/login", request.url));
  }
  
  // Se tem sessão e tenta acessar login
  if (session && request.nextUrl.pathname === "/login") {
    return NextResponse.redirect(new URL("/painel", request.url));
  }
  
  return NextResponse.next();
}
